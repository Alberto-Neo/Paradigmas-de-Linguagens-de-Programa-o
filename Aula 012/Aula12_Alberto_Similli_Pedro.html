na recursão comum, a pilha de chamadas cresce porque existem operações esperando o retorno das funções. na recursão em cauda, o Scheme consegue reutilizar o espaço das chamadas.

(define (tamanho lis)
  (if (null? lis)
   0
   (+ 1 (tamanho (cdr lis)))))

(define (soma-quadrados lis)
  (if (null? lis)
      0
      (+ (* (car lis) (car lis))
         (soma-quadrados (cdr lis)))))

(define (pares lis)
  (cond
    ((null? lis) '())
    ((even? (car lis))
     (cons (car lis) (pares (cdr lis))))
    (else (pares (cdr lis)))))
